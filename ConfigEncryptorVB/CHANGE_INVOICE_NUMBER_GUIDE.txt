1️⃣ فواتير البيع (Gold Sell):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   الجداول:
   • BAY → BAY_SARYAL (رقم الفاتورة)
   • selInvoceDetels → invoc_time (رقم الفاتورة)
   
   العلاقة:
   BAY.BAY_SARYAL = selInvoceDetels.invoc_time


2️⃣ فواتير الشراء (Gold Buy):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   الجداول:
   • bay_goled_mony → saryal (رقم الفاتورة)
   • Buy_Gold → inv_id (رقم الفاتورة)
   
   العلاقة:
   bay_goled_mony.saryal = Buy_Gold.inv_id

الخطوة 2️⃣: اختيار نوع الفاتورة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
○ فاتورة بيع - Sell Invoice
○ فاتورة شراء - Buy Invoice


الخطوة 3️⃣: إدخال رقم الفاتورة القديم والبحث
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. أدخل رقم الفاتورة القديم
2. اضغط "🔍 بحث - Search"
3. سيتم عرض تفاصيل الفاتورة الكاملة:
   • بيانات الفاتورة
   • تفاصيل الأصناف/القطع
   • المبالغ
   • التواريخ
   • الحالة (نشطة/محذوفة)


الخطوة 4️⃣: إدخال رقم الفاتورة الجديد
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. أدخل رقم الفاتورة الجديد
2. اضغط "💾 حفظ - Save"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

5. Rollback (إذا فشل)

═══════════════════════════════════════════════════════════════
                    🔒 الأمان والحماية
═══════════════════════════════════════════════════════════════

1️⃣ عرض تفاصيل الفاتورة الكاملة قبل التغيير
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • يمكن مراجعة جميع البيانات قبل التأكيد
   • يتأكد المستخدم من اختيار الفاتورة الصحيحة


2️⃣ التأكيد المزدوج
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • تأكيد نعم/لا
   • كلمة تأكيد "EDIT"
   • يمنع التنفيذ العرضي


3️⃣ استخدام Transaction
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • إذا فشل تحديث أي جدول → يتم التراجع عن الكل
   • لا يمكن حدوث تحديث جزئي
   • البيانات تبقى متناسقة


4️⃣ استخدام N'...' للـ Unicode
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   • يدعم أرقام الفواتير الكبيرة (nvarchar)
   • لا يوجد overflow
   • يدعم الأحرف العربية إذا لزم الأمر

═══════════════════════════════════════════════════════════════
                    ⚠️ تحذيرات هامة
═══════════════════════════════════════════════════════════════

1. ⚠️ لا يمكن التراجع عن التغيير بعد التأكيد
   • تأكد من رقم الفاتورة الجديد صحيح
   • راجع التفاصيل جيداً قبل الحفظ

2. ⚠️ تأكد من عدم وجود فاتورة برقم الجديد مسبقاً
   • قد يسبب تضارب في البيانات
   • يُفضل التحقق أولاً

3. ⚠️ يجب عمل Backup قبل التغيير
   • احتياطاً من أي مشكلة
   • للرجوع إذا حدث خطأ
